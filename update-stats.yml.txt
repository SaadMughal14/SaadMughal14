name: Update Live Stats

on:
  schedule:
    - cron: "0 4 * * *"   # daily at 04:00 UTC
  workflow_dispatch:
  push:
    branches: [main]
    paths-ignore:
      - "stats.svg"
      - "langs.svg"
      - "trophies.svg"

permissions:
  contents: write

jobs:
  update-stats:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Install dependencies
        run: pip install requests

      - name: Generate live stats SVGs
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          STATS_USERNAME: ${{ github.repository_owner }}
        run: python scripts/generate_stats.py

      - name: Commit changes if any
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add stats.svg langs.svg trophies.svg
          git diff --staged --quiet || git commit -m "chore: update live GitHub stats"
          git push
